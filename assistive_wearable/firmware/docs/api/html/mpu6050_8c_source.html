109 sum_g[axis] += (float)gv / 32.8f;

116 }
117 gravity_mag = sqrtf(gravity_lpf[0] * gravity_lpf[0] + gravity_lpf[1] * gravity_lpf[1] + gravity_lpf[2] * gravity_lpf[2]);
118 ESP_LOGI(TAG, "calibration complete: gyro bias %.2f %.2f %.2f dps", gyro_bias[0], gyro_bias[1], gyro_bias[2]);
